// ==UserScript==
// @name         Vikunja Kiosk Helper
// @namespace    http://tampermonkey.net/
// @version      1.3
// @description  Helper that fade in/out at reload and set class for overdue task
// @match        https://your.vikunja.url.here/*
// @grant        none
// ==/UserScript==

(function() {
    'use strict';

    // Avoid initializing multiple times (e.g., when the script is injected repeatedly)
    if (window.vikunjaKioskHelperInitialized) {
        return;
    }
    window.vikunjaKioskHelperInitialized = true;

// 1. set opacity and start timer with inline-style
    if (!document.getElementById('kiosk-fade-style')) {
        const style = document.createElement('style');
        style.id = 'kiosk-fade-style';
        style.innerHTML = `
            body {
                opacity: 0 !important;
                transition: opacity 0.8s ease-in-out !important;
            }
        `;
        document.head.appendChild(style);
    }

    // 2. fade back in after 2 second, should be long enough to bypass the "login" screen
    setTimeout(() => {
        const styleTag = document.getElementById('kiosk-fade-style');
        if (styleTag) {
            styleTag.remove(); // clear opacity
        }
        document.body.style.opacity = '1';
    }, 2000); // 2000 milliseconds = 2 seconds

    // 3. helper for marking overdue tasks; sets due tasks with "before X days" with new class .is-overdue
    // may have to be set to your selected language, in english its "ago", in german its "vor", both included already
    function markOverdueTasks() {
        document.querySelectorAll('[class*="dueDate"]').forEach(el => {
            const text = el.textContent.toLowerCase();
                if (text.includes('vor') || text.includes('ago')) {
                    el.classList.add('is-overdue');
                } else {
                el.classList.remove('is-overdue');
            }
        });
    }

    // Run marking once because the page doesn't auto-update — avoid continuous polling
    function scheduleMarkOverdue() {
        // try after DOMContentLoaded and a short delay to allow Vikunja to render
        if (document.readyState === 'complete' || document.readyState === 'interactive') {
            setTimeout(markOverdueTasks, 800);
        } else {
            document.addEventListener('DOMContentLoaded', () => setTimeout(markOverdueTasks, 800), { once: true });
        }
        // safety call in case content renders slightly later
        setTimeout(markOverdueTasks, 1600);
    }
    scheduleMarkOverdue();

    // 4. make a quick fade-out on leave/reload the page
    window.addEventListener('beforeunload', function () {
        document.body.style.transition = 'opacity 0.2s ease-in-out';
        document.body.style.opacity = '0';
    });

    // 5. autoscroll feature
    // ==========================================
    // settings for autoscroll
    // ==========================================
    const scrollSpeed = 1; // pixel per step (1 = slow & smooth)
    const scrollInterval = 40; // delay per step in ms (40ms = smooth)
    const startDelay = 5000; // delay for scroll to start (5 sec)
    const pauseAtBottom = 3000; // delay to scroll back up  (3 sec)
    // ==========================================

    let isScrolling = false;

    function startAutoScroll() {
        if (window.vikunjaAutoScrollRunning) return;
        window.vikunjaAutoScrollRunning = true;

        // clear any existing interval handle
        if (window.vikunjaAutoScrollInterval) {
            clearInterval(window.vikunjaAutoScrollInterval);
            window.vikunjaAutoScrollInterval = null;
        }

        // clear any previous bottom timeout/poll
        if (window.vikunjaAutoScrollBottomTimeout) {
            clearTimeout(window.vikunjaAutoScrollBottomTimeout);
            window.vikunjaAutoScrollBottomTimeout = null;
        }
        if (window.vikunjaScrollTopPoll) {
            clearInterval(window.vikunjaScrollTopPoll);
            window.vikunjaScrollTopPoll = null;
        }

        window.vikunjaAutoScrollInterval = setInterval(() => {
            const reachedBottom = (window.innerHeight + window.scrollY) >= (document.documentElement.scrollHeight - 2);

            if (!reachedBottom) {
                window.scrollBy(0, scrollSpeed);
                return;
            }

            // reached bottom -> stop the scrolling interval
            clearInterval(window.vikunjaAutoScrollInterval);
            window.vikunjaAutoScrollInterval = null;

            // wait at bottom for pauseAtBottom ms, then smoothly scroll to top and wait until at top
            window.vikunjaAutoScrollBottomTimeout = setTimeout(() => {
                try {
                    window.scrollTo({ top: 0, behavior: 'smooth' });
                } catch (e) {
                    // fallback
                    window.scrollTo(0, 0);
                }

                // poll until we've reached the top (or until timeout)
                const pollInterval = 150; // ms
                let pollCount = 0;
                const maxPolls = Math.ceil(15000 / pollInterval); // ~15s safety

                window.vikunjaScrollTopPoll = setInterval(() => {
                    pollCount++;
                    const atTop = window.scrollY === 0;
                    if (atTop || pollCount >= maxPolls) {
                        clearInterval(window.vikunjaScrollTopPoll);
                        window.vikunjaScrollTopPoll = null;

                        // allow a short pause at top, then restart scrolling cycle
                        window.vikunjaAutoScrollRunning = false;
                        if (window.vikunjaAutoScrollStartTimeout) clearTimeout(window.vikunjaAutoScrollStartTimeout);
                        window.vikunjaAutoScrollStartTimeout = setTimeout(() => {
                            startAutoScroll();
                        }, pauseAtBottom);
                    }
                }, pollInterval);
            }, pauseAtBottom);
        }, scrollInterval);
    }

    // schedule start but avoid duplicate timers
    if (window.vikunjaAutoScrollStartTimeout) {
        clearTimeout(window.vikunjaAutoScrollStartTimeout);
    }
    window.vikunjaAutoScrollStartTimeout = setTimeout(startAutoScroll, startDelay);
   
})();
